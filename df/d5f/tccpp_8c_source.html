68{
69 TokenSym *ts, **ptable;

71

114static TokenSym *tok_alloc(const char *str, int len)
115{
116 TokenSym *ts, **pts;
117 int i;
118 unsigned int h;
119
120 h = TOK_HASH_INIT;
121 for (i = 0; i < len; i++)
122 h = TOK_HASH_FUNC(h, ((unsigned char *) str)[i]);
123 h &= (TOK_HASH_SIZE - 1);
124
125 pts = &hash_ident[h];
126 for (;;) {
127 ts = *pts;
128 if (!ts)
129 break;
130 if (ts->len == len && !memcmp(ts->str, str, len))
131 return ts;
132 pts = &(ts->hash_next);
133 }
134 return tok_alloc_new(pts, str, len);
135}

149char *get_tok_str(int v, CValue *cv)
150{
151 static char buf[STRING_MAX_SIZE + 1];
152 static CString cstr_buf;
153 CString *cstr;
154 CValue cval;
155 char *p;
156 int i, len;
157
158 if (!cv) {
159 cval.ull = 0;
160 cv = &cval;
161 }
162
163 /* NOTE: to go faster, we give a fixed buffer for small strings */
164 cstr_reset(&cstr_buf);
165 cstr_buf.data = buf;
166 cstr_buf.size_allocated = sizeof(buf);
167 p = buf;
168
169 switch (v) {
170 case TOK_CINT:
171 case TOK_CUINT:
172 /* XXX: not quite exact, but only useful for testing */
173 sprintf(p, "%u", cv->ui);
174 break;
175 case TOK_CLLONG:
176 case TOK_CULLONG:
177 /* XXX: not quite exact, but only useful for testing */
178#ifdef _WIN32
179 sprintf(p, "%u", (unsigned) cv->ull);
180#else
181 sprintf(p, "%Lu", cv->ull);
182#endif
183 break;
184 case TOK_LCHAR:
185 cstr_ccat(&cstr_buf, 'L');
186 case TOK_CCHAR:
187 cstr_ccat(&cstr_buf, '\'');
188 add_char(&cstr_buf, cv->i);
189 cstr_ccat(&cstr_buf, '\'');
190 cstr_ccat(&cstr_buf, '\0');
191 break;
192 case TOK_PPNUM:
193 cstr = cv->cstr;
194 len = cstr->size - 1;
195 for (i = 0; i < len; i++)
196 add_char(&cstr_buf, ((unsigned char *) cstr->data)[i]);
197 cstr_ccat(&cstr_buf, '\0');
198 break;
199 case TOK_LSTR:
200 cstr_ccat(&cstr_buf, 'L');
201 case TOK_STR:
202 cstr = cv->cstr;
203 cstr_ccat(&cstr_buf, '\"');
204 if (v == TOK_STR) {
205 len = cstr->size - 1;
206 for (i = 0; i < len; i++)
207 add_char(&cstr_buf, ((unsigned char *) cstr->data)[i]);
208 } else {
209 len = (cstr->size / sizeof(nwchar_t)) - 1;
210 for (i = 0; i < len; i++)
211 add_char(&cstr_buf, ((nwchar_t *) cstr->data)[i]);
212 }
213 cstr_ccat(&cstr_buf, '\"');
214 cstr_ccat(&cstr_buf, '\0');
215 break;
216 case TOK_LT:
217 v = '<';
218 goto addv;
219 case TOK_GT:
220 v = '>';
221 goto addv;
222 case TOK_DOTS:
223 return strcpy(p, "...");
224 case TOK_A_SHL:
225 return strcpy(p, "<<=");
226 case TOK_A_SAR:
227 return strcpy(p, ">>=");
228 default:
229 if (v < TOK_IDENT) {
230 /* search in two bytes table */
231 const unsigned char *q = tok_two_chars;
232 while (*q) {
233 if (q[2] == v) {
234 *p++ = q[0];
235 *p++ = q[1];
236 *p = '\0';
237 return buf;
238 }
239 q += 3;
240 }
241 addv:
242 *p++ = v;
243 *p = '\0';
244 } else if (v < tok_ident) {
245 return table_ident[v - TOK_IDENT]->str;
246 } else if (v >= SYM_FIRST_ANOM) {
247 /* special name for anonymous symbol */
248#ifdef TCC_TARGET_816
249 sprintf(p, "L.%s%d", unique_token, v - SYM_FIRST_ANOM);
250#else
251 sprintf(p, "L.%u", v - SYM_FIRST_ANOM);
252#endif
253 } else {
254 /* should never happen */
255 return NULL;
256 }
257 break;
258 }
259 return cstr_buf.data;
260}

500static uint8_t *parse_comment(uint8_t *p)
501{
502 int c;
503
504 p++;
505 for (;;) {
506 /* fast skip loop */
507 for (;;) {
508 c = *p;
509 if (c == '\n' || c == '*' || c == '\\')
510 break;
511 p++;
512 c = *p;
513 if (c == '\n' || c == '*' || c == '\\')
514 break;
515 p++;
516 }
517 /* now we can handle all the cases */
518 if (c == '\n') {
519 file->line_num++;
520 p++;
521 } else if (c == '*') {
522 p++;
523 for (;;) {
524 c = *p;
525 if (c == '*') {
526 p++;
527 } else if (c == '/') {
528 goto end_of_comment;
529 } else if (c == '\\') {
530 file->buf_ptr = p;
531 c = handle_eob();
532 p = file->buf_ptr;
533 if (c == '\\') {
534 /* skip '\[\r]\n', otherwise just skip the stray */
535 while (c == '\\') {
536 PEEKC_EOB(c, p);
537 if (c == '\n') {
538 file->line_num++;
539 PEEKC_EOB(c, p);
540 } else if (c == '\r') {
541 PEEKC_EOB(c, p);
542 if (c == '\n') {
543 file->line_num++;
544 PEEKC_EOB(c, p);
545 }
546 } else {
547 goto after_star;
548 }
549 }
550 }
551 } else {
552 break;
553 }
554 }
555 after_star:;
556 } else {
557 /* stray, eob or eof */
558 file->buf_ptr = p;
559 c = handle_eob();
560 p = file->buf_ptr;
561 if (c == CH_EOF) {
562 error("unexpected end of file in comment");
563 } else if (c == '\\') {
564 p++;
565 }
566 }
567 }
568end_of_comment:
569 p++;
570 return p;
571}

623static uint8_t *parse_pp_string(uint8_t *p, int sep, CString *str)
624{
625 int c;
626 p++;
627 for (;;) {
628 c = *p;
629 if (c == sep) {
630 break;
631 } else if (c == '\\') {
632 file->buf_ptr = p;
633 c = handle_eob();
634 p = file->buf_ptr;
635 if (c == CH_EOF) {
636 unterminated_string:
637 /* XXX: indicate line number of start of string */
638 error("missing terminating %c character", sep);
639 } else if (c == '\\') {
640 /* escape : just skip \[\r]\n */
641 PEEKC_EOB(c, p);
642 if (c == '\n') {
643 file->line_num++;
644 p++;
645 } else if (c == '\r') {
646 PEEKC_EOB(c, p);
647 if (c != '\n')
648 expect("'\n' after '\r'");
649 file->line_num++;
650 p++;
651 } else if (c == CH_EOF) {
652 goto unterminated_string;
653 } else {
654 if (str) {
655 cstr_ccat(str, '\\');
656 cstr_ccat(str, c);
657 }
658 p++;
659 }
660 }
661 } else if (c == '\n') {
662 file->line_num++;
663 goto add_char;
664 } else if (c == '\r') {
665 PEEKC_EOB(c, p);
666 if (c != '\n') {
667 if (str)
668 cstr_ccat(str, '\r');
669 } else {
670 file->line_num++;
671 goto add_char;
672 }
673 } else {
674 add_char:
675 if (str)
676 cstr_ccat(str, c);
677 p++;
678 }
679 }
680 p++;
681 return p;
682}

691void preprocess_skip(void)
692{
693 int a, start_of_line, c, in_warn_or_error;
694 uint8_t *p;
695
696 p = file->buf_ptr;
697 a = 0;
698redo_start:
699 start_of_line = 1;
700 in_warn_or_error = 0;
701 for (;;) {
702 redo_no_start:
703 c = *p;
704 switch (c) {
705 case ' ':
706 case '\t':
707 case '\f':
708 case '\v':
709 case '\r':
710 p++;
711 goto redo_no_start;
712 case '\n':
713 file->line_num++;
714 p++;
715 goto redo_start;
716 case '\\':
717 file->buf_ptr = p;
718 c = handle_eob();
719 if (c == CH_EOF) {
720 expect("#endif");
721 } else if (c == '\\') {
722 ch = file->buf_ptr[0];
723 handle_stray_noerror();
724 }
725 p = file->buf_ptr;
726 goto redo_no_start;
727 /* skip strings */
728 case '\"':
729 case '\'':
730 if (in_warn_or_error)
731 goto _default;
732 p = parse_pp_string(p, c, NULL);
733 break;
734 /* skip comments */
735 case '/':
736 if (in_warn_or_error)
737 goto _default;
738 file->buf_ptr = p;
739 ch = *p;
740 minp();
741 p = file->buf_ptr;
742 if (ch == '*') {
743 p = parse_comment(p);
744 } else if (ch == '/') {
745 p = parse_line_comment(p);
746 }
747 break;
748 case '#':
749 p++;
750 if (start_of_line) {
751 file->buf_ptr = p;
752 next_nomacro();
753 p = file->buf_ptr;
754 if (a == 0 && (tok == TOK_ELSE || tok == TOK_ELIF || tok == TOK_ENDIF))
755 goto the_end;
756 if (tok == TOK_IF || tok == TOK_IFDEF || tok == TOK_IFNDEF)
757 a++;
758 else if (tok == TOK_ENDIF)
759 a--;
760 else if (tok == TOK_ERROR || tok == TOK_WARNING)
761 in_warn_or_error = 1;
762 else if (tok == TOK_LINEFEED)
763 goto redo_start;
764 }
765 break;
766 _default:
767 default:
768 p++;
769 break;
770 }
771 start_of_line = 0;
772 }
773the_end:;
774 file->buf_ptr = p;
775}

943static void tok_str_add2(TokenString *s, int t, CValue *cv)
944{
945 int len, *str;
946
947 len = s->len;
948 str = s->str;
949
950 /* allocate space for worst case */
951 if (len + TOK_MAX_SIZE > s->allocated_len)
952 str = tok_str_realloc(s);
953 str[len++] = t;
954 switch (t) {
955 case TOK_CINT:
956 case TOK_CUINT:
957 case TOK_CCHAR:
958 case TOK_LCHAR:
959 case TOK_CFLOAT:
960#ifdef TCC_TARGET_816

964 str[len++] = cv->tab[0];
965 break;
966 case TOK_PPNUM:
967 case TOK_STR:
968 case TOK_LSTR: {
969 int nb_words;
970 CString *cstr;
971

973 while ((len + nb_words) > s->allocated_len)
974 str = tok_str_realloc(s);

976 cstr->data = NULL;
977 cstr->size = cv->cstr->size;
978 cstr->data_allocated = NULL;
979 cstr->size_allocated = cstr->size;
980 memcpy((char *) cstr + sizeof(CString), cv->cstr->data, cstr->size);
981 len += nb_words;
982 } break;
983#ifndef TCC_TARGET_816
984 case TOK_CDOUBLE:
985#endif
986 case TOK_CLLONG:
987 case TOK_CULLONG:
988#if LDOUBLE_SIZE == 8
989 case TOK_CLDOUBLE:
990#endif
991 str[len++] = cv->tab[0];
992 str[len++] = cv->tab[1];
993 break;
994#if LDOUBLE_SIZE == 12
995 case TOK_CLDOUBLE:
996 str[len++] = cv->tab[0];

998 str[len++] = cv->tab[2];
999#elif LDOUBLE_SIZE == 16

1002 str[len++] = cv->tab[1];
1003 str[len++] = cv->tab[2];
1004 str[len++] = cv->tab[3];
1005#elif LDOUBLE_SIZE != 8
1006#error add long double size support
1007#endif
1008 break;
1009 default:
1010 break;
1011 }
1012 s->len = len;

1286/* parse after #define */
1287static void parse_define(void)
1288{
1289 Sym *s, *first, **ps;
1290 int v, t, varg, is_vaargs, spc;
1291 TokenString str;
1292
1293 v = tok;
1294 if (v < TOK_IDENT)
1295 error("invalid macro name '%s'", get_tok_str(tok, &tokc));
1296 /* XXX: should check if same macro (ANSI) */
1297 first = NULL;
1298 t = MACRO_OBJ;
1299 /* '(' must be just after macro definition for MACRO_FUNC */
1300 next_nomacro_spc();
1301 if (tok == '(') {
1302 next_nomacro();
1303 ps = &first;
1304 while (tok != ')') {
1305 varg = tok;
1306 next_nomacro();
1307 is_vaargs = 0;
1308 if (varg == TOK_DOTS) {
1309 varg = TOK___VA_ARGS__;
1310 is_vaargs = 1;
1311 } else if (tok == TOK_DOTS && gnu_ext) {
1312 is_vaargs = 1;
1313 next_nomacro();
1314 }
1315 if (varg < TOK_IDENT)
1316 error("badly punctuated parameter list");
1317 s = sym_push2(&define_stack, varg | SYM_FIELD, is_vaargs, 0);
1318 *ps = s;
1319 ps = &s->next;
1320 if (tok != ',')
1321 break;
1322 next_nomacro();
1323 }
1324 if (tok == ')')
1325 next_nomacro_spc();
1326 t = MACRO_FUNC;
1327 }
1328 tok_str_new(&str);
1329 spc = 2;
1330 /* EOF testing necessary for '-D' handling */
1331 while (tok != TOK_LINEFEED && tok != TOK_EOF) {
1332 /* remove spaces around ## and after '#' */
1333 if (TOK_TWOSHARPS == tok) {
1334 if (1 == spc)
1335 --str.len;
1336 spc = 2;
1337 } else if ('#' == tok) {
1338 spc = 2;
1339 } else if (check_space(tok, &spc)) {
1340 goto skip;
1341 }
1342 tok_str_add2(&str, tok, &tokc);
1343 skip:
1344 next_nomacro_spc();
1345 }
1346 if (spc == 1)
1347 --str.len; /* remove trailing space */
1348 tok_str_add(&str, 0);
1349#ifdef PP_DEBUG
1350 printf("define %s %d: ", get_tok_str(v, NULL), t);
1351 tok_print(str.str);
1352#endif
1353 define_push(v, t, str.str, first);
1354}

1413static void pragma_parse(TCCState *s1)
1414{
1415 int val;
1416
1417 next();
1418 if (tok == TOK_pack) {
1419 /*
1420 This may be:
1421 #pragma pack(1) // set
1422 #pragma pack() // reset to default
1423 #pragma pack(push,1) // push & set
1424 #pragma pack(pop) // restore previous
1425 */
1426 next();
1427 skip('(');
1428 if (tok == TOK_ASM_pop) {
1429 next();
1430 if (s1->pack_stack_ptr <= s1->pack_stack) {
1431 stk_error:
1432 error("out of pack stack");
1433 }
1434 s1->pack_stack_ptr--;
1435 } else {
1436 val = 0;
1437 if (tok != ')') {
1438 if (tok == TOK_ASM_push) {
1439 next();
1440 if (s1->pack_stack_ptr >= s1->pack_stack + PACK_STACK_SIZE - 1)
1441 goto stk_error;
1442 s1->pack_stack_ptr++;
1443 skip(',');
1444 }
1445 if (tok != TOK_CINT) {
1446 pack_error:
1447 error("invalid pack pragma");
1448 }
1449 val = tokc.i;
1450 if (val < 1 || val > 16 || (val & (val - 1)) != 0)
1451 goto pack_error;
1452 next();
1453 }
1454 *s1->pack_stack_ptr = val;
1455 skip(')');
1456 }
1457 }
1458}

1460/* is_bof is true if first non space token at beginning of file */
1461static void preprocess(int is_bof)
1462{
1463 TCCState *s1 = tcc_state;
1464 int i, c, n, saved_parse_flags;
1465 char buf[1024], *q;
1466 Sym *s;
1467
1468 saved_parse_flags = parse_flags;
1469 parse_flags = PARSE_FLAG_PREPROCESS | PARSE_FLAG_TOK_NUM | PARSE_FLAG_LINEFEED;
1470 next_nomacro();
1471redo:
1472 switch (tok) {
1473 case TOK_DEFINE:
1474 next_nomacro();
1475 parse_define();
1476 break;
1477 case TOK_UNDEF:
1478 next_nomacro();
1479 s = define_find(tok);
1480 /* undefine symbol by putting an invalid name */
1481 if (s)
1482 define_undef(s);
1483 break;
1484 case TOK_INCLUDE:
1485 case TOK_INCLUDE_NEXT:
1486 ch = file->buf_ptr[0];
1487 /* XXX: incorrect if comments : use next_nomacro with a special mode */
1488 skip_spaces();
1489 if (ch == '<') {
1490 c = '>';
1491 goto read_name;
1492 } else if (ch == '\"') {
1493 c = ch;
1494 read_name:
1495 inp();
1496 q = buf;
1497 while (ch != c && ch != '\n' && ch != CH_EOF) {
1498 if ((q - buf) < sizeof(buf) - 1)
1499 *q++ = ch;
1500 if (ch == '\\') {
1501 if (handle_stray_noerror() == 0)
1502 --q;
1503 } else
1504 inp();
1505 }
1506 *q = '\0';
1507 minp();
1508#if 0
1509 /* eat all spaces and comments after include */
1510 /* XXX: slightly incorrect */
1511 while (ch1 != '\n' && ch1 != CH_EOF)
1512 inp();
1513#endif
1514 } else {
1515 /* computed #include : either we have only strings or
1516 we have anything enclosed in '<>' */
1517 next();
1518 buf[0] = '\0';
1519 if (tok == TOK_STR) {
1520 while (tok != TOK_LINEFEED) {
1521 if (tok != TOK_STR) {
1522 include_syntax:
1523 error("'#include' expects \"FILENAME\" or <FILENAME>");
1524 }
1525 pstrcat(buf, sizeof(buf), (char *) tokc.cstr->data);
1526 next();
1527 }
1528 c = '\"';
1529 } else {
1530 int len;
1531 while (tok != TOK_LINEFEED) {
1532 pstrcat(buf, sizeof(buf), get_tok_str(tok, &tokc));
1533 next();
1534 }
1535 len = strlen(buf);
1536 /* check syntax and remove '<>' */
1537 if (len < 2 || buf[0] != '<' || buf[len - 1] != '>')
1538 goto include_syntax;
1539 memmove(buf, buf + 1, len - 2);
1540 buf[len - 2] = '\0';
1541 c = '>';
1542 }
1543 }
1544
1545 if (s1->include_stack_ptr >= s1->include_stack + INCLUDE_STACK_SIZE)
1546 error("#include recursion too deep");
1547
1548 n = s1->nb_include_paths + s1->nb_sysinclude_paths;
1549 for (i = -2; i < n; ++i) {
1550 char buf1[sizeof file->filename];
1551 BufferedFile *f;
1552 CachedInclude *e;
1553 const char *path;
1554 int size;
1555
1556 if (i == -2) {
1557 /* check absolute include path */
1558 if (!IS_ABSPATH(buf))
1559 continue;
1560 buf1[0] = 0;
1561
1562 } else if (i == -1) {
1563 /* search in current dir if "header.h" */
1564 if (c != '\"')
1565 continue;
1566 size = tcc_basename(file->filename) - file->filename;
1567 memcpy(buf1, file->filename, size);
1568 buf1[size] = '\0';
1569
1570 } else {
1571 /* search in all the include paths */
1572 if (i < s1->nb_include_paths)
1573 path = s1->include_paths[i];
1574 else
1575 path = s1->sysinclude_paths[i - s1->nb_include_paths];
1576 pstrcpy(buf1, sizeof(buf1), path);
1577 pstrcat(buf1, sizeof(buf1), "/");
1578 }
1579
1580 pstrcat(buf1, sizeof(buf1), buf);
1581
1582 e = search_cached_include(s1, c, buf1);
1583 if (e && define_find(e->ifndef_macro)) {
1584 /* no need to parse the include because the 'ifndef macro'
1585 is defined */
1586#ifdef INC_DEBUG
1587 printf("%s: skipping %s\n", file->filename, buf);
1588#endif
1589 f = NULL;
1590 } else {
1591 f = tcc_open(s1, buf1);
1592 if (!f)
1593 continue;
1594 }
1595
1596 if (tok == TOK_INCLUDE_NEXT) {
1597 tok = TOK_INCLUDE;
1598 if (f)
1599 tcc_close(f);
1600 continue;
1601 }
1602
1603 if (!f)
1604 goto include_done;
1605
1606#ifdef INC_DEBUG
1607 printf("%s: including %s\n", file->filename, buf1);
1608#endif
1609
1610 /* XXX: fix current line init */
1611 /* push current file in stack */
1612 *s1->include_stack_ptr++ = file;
1613 f->inc_type = c;
1614 pstrcpy(f->inc_filename, sizeof(f->inc_filename), buf1);
1615 file = f;
1616 /* add include file debug info */
1617 if (tcc_state->do_debug) {
1618 put_stabs(file->filename, N_BINCL, 0, 0, 0);
1619 }
1620 tok_flags |= TOK_FLAG_BOF | TOK_FLAG_BOL;
1621 ch = file->buf_ptr[0];
1622 goto the_end;
1623 }
1624 error("include file '%s' not found", buf);
1625 include_done:
1626 break;
1627 case TOK_IFNDEF:
1628 c = 1;
1629 goto do_ifdef;
1630 case TOK_IF:
1631 c = expr_preprocess();
1632 goto do_if;
1633 case TOK_IFDEF:
1634 c = 0;
1635 do_ifdef:
1636 next_nomacro();
1637 if (tok < TOK_IDENT)
1638 error("invalid argument for '#if%sdef'", c ? "n" : "");
1639 if (is_bof) {
1640 if (c) {
1641#ifdef INC_DEBUG
1642 printf("#ifndef %s\n", get_tok_str(tok, NULL));
1643#endif
1644 file->ifndef_macro = tok;
1645 }
1646 }
1647 c = (define_find(tok) != 0) ^ c;
1648 do_if:
1649 if (s1->ifdef_stack_ptr >= s1->ifdef_stack + IFDEF_STACK_SIZE)
1650 error("memory full");
1651 *s1->ifdef_stack_ptr++ = c;
1652 goto test_skip;
1653 case TOK_ELSE:
1654 if (s1->ifdef_stack_ptr == s1->ifdef_stack)
1655 error("#else without matching #if");
1656 if (s1->ifdef_stack_ptr[-1] & 2)
1657 error("#else after #else");
1658 c = (s1->ifdef_stack_ptr[-1] ^= 3);
1659 goto test_else;
1660 case TOK_ELIF:
1661 if (s1->ifdef_stack_ptr == s1->ifdef_stack)
1662 error("#elif without matching #if");
1663 c = s1->ifdef_stack_ptr[-1];
1664 if (c > 1)
1665 error("#elif after #else");
1666 /* last #if/#elif expression was true: we skip */
1667 if (c == 1)
1668 goto skip;
1669 c = expr_preprocess();
1670 s1->ifdef_stack_ptr[-1] = c;
1671 test_else:
1672 if (s1->ifdef_stack_ptr == file->ifdef_stack_ptr + 1)
1673 file->ifndef_macro = 0;
1674 test_skip:
1675 if (!(c & 1)) {
1676 skip:
1677 preprocess_skip();
1678 is_bof = 0;
1679 goto redo;
1680 }
1681 break;
1682 case TOK_ENDIF:
1683 if (s1->ifdef_stack_ptr <= file->ifdef_stack_ptr)
1684 error("#endif without matching #if");
1685 s1->ifdef_stack_ptr--;
1686 /* '#ifndef macro' was at the start of file. Now we check if
1687 an '#endif' is exactly at the end of file */
1688 if (file->ifndef_macro && s1->ifdef_stack_ptr == file->ifdef_stack_ptr) {
1689 file->ifndef_macro_saved = file->ifndef_macro;
1690 /* need to set to zero to avoid false matches if another
1691 #ifndef at middle of file */
1692 file->ifndef_macro = 0;
1693 while (tok != TOK_LINEFEED)
1694 next_nomacro();
1695 tok_flags |= TOK_FLAG_ENDIF;
1696 goto the_end;
1697 }
1698 break;
1699 case TOK_LINE:
1700 next();
1701 if (tok != TOK_CINT)
1702 error("#line");
1703 file->line_num = tokc.i - 1; /* the line number will be incremented after */
1704 next();
1705 if (tok != TOK_LINEFEED) {
1706 if (tok != TOK_STR)
1707 error("#line");
1708 pstrcpy(file->filename, sizeof(file->filename), (char *) tokc.cstr->data);
1709 }
1710 break;
1711 case TOK_ERROR:
1712 case TOK_WARNING:
1713 c = tok;
1714 ch = file->buf_ptr[0];
1715 skip_spaces();
1716 q = buf;
1717 while (ch != '\n' && ch != CH_EOF) {
1718 if ((q - buf) < sizeof(buf) - 1)
1719 *q++ = ch;
1720 if (ch == '\\') {
1721 if (handle_stray_noerror() == 0)
1722 --q;
1723 } else
1724 inp();
1725 }
1726 *q = '\0';
1727 if (c == TOK_ERROR)
1728 error("#error %s", buf);
1729 else
1730 warning("#warning %s", buf);
1731 break;
1732 case TOK_PRAGMA:
1733 pragma_parse(s1);
1734 break;
1735 default:
1736 if (tok == TOK_LINEFEED || tok == '!' || tok == TOK_PPNUM) {
1737 /* '!' is ignored to allow C scripts. numbers are ignored
1738 to emulate cpp behaviour */
1739 } else {
1740 if (!(saved_parse_flags & PARSE_FLAG_ASM_COMMENTS))
1741 warning("Ignoring unknown preprocessing directive #%s", get_tok_str(tok, &tokc));
1742 else {
1743 /* this is a gas line comment in an 'S' file. */
1744 file->buf_ptr = parse_line_comment(file->buf_ptr);
1745 goto the_end;
1746 }
1747 }
1748 break;
1749 }
1750 /* ignore other preprocess commands or #! for C scripts */
1751 while (tok != TOK_LINEFEED)
1752 next_nomacro();
1753the_end:
1754 parse_flags = saved_parse_flags;
1755}

1757/* evaluate escape codes in a string. */
1758static void parse_escape_string(CString *outstr, const uint8_t *buf, int is_long)
1759{
1760 int c, n;
1761 const uint8_t *p;
1762
1763 p = buf;
1764 for (;;) {
1765 c = *p;
1766 if (c == '\0')
1767 break;
1768 if (c == '\\') {
1769 p++;
1770 /* escape */
1771 c = *p;
1772 switch (c) {
1773 case '0':
1774 case '1':
1775 case '2':
1776 case '3':
1777 case '4':
1778 case '5':
1779 case '6':
1780 case '7':
1781 /* at most three octal digits */
1782 n = c - '0';
1783 p++;
1784 c = *p;
1785 if (isoct(c)) {
1786 n = n * 8 + c - '0';
1787 p++;
1788 c = *p;
1789 if (isoct(c)) {
1790 n = n * 8 + c - '0';

1821 break;
1822 case 'f':

1824 break;
1825 case 'n':

1827 break;

1833 break;

1836 break;
1837 case 'e':

1874void bn_lshift(unsigned int *bn, int shift, int or_val)
1875{

1878 for (i = 0; i < BN_SIZE; i++) {

1881 or_val = v >> (32 - shift);
1882 }
