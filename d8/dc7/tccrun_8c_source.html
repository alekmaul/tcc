34/* Do all relocations (needed before using tcc_get_symbol())
35 Returns -1 on error. */
36
37int tcc_relocate(TCCState *s1)
38{
39 int ret;
40
41 ret = tcc_relocate_ex(s1, NULL);
42 if (-1 != ret) {
43 s1->runtime_mem = tcc_malloc(ret);
44 ret = tcc_relocate_ex(s1, s1->runtime_mem);
45 }
46 return ret;
47}

49/* launch the compiled program with the given arguments */
50int tcc_run(TCCState *s1, int argc, char **argv)
51{
52 int (*prog_main)(int, char **);
53
54 if (tcc_relocate(s1) < 0)
55 return -1;
56
57 prog_main = tcc_get_symbol_err(s1, "main");
58
59#ifdef CONFIG_TCC_BACKTRACE
60 if (s1->do_debug)
61 set_exception_handler();
62#endif
63
64#ifdef CONFIG_TCC_BCHECK
65 if (s1->do_bounds_check) {
66 void (*bound_init)(void);
67 void (*bound_exit)(void);
68 int ret;
69 /* set error function */
70 rt_bound_error_msg = tcc_get_symbol_err(s1, "__bound_error_msg");
71 rt_prog_main = prog_main;
72 /* XXX: use .init section so that it also work in binary ? */
73 bound_init = tcc_get_symbol_err(s1, "__bound_init");
74 bound_exit = tcc_get_symbol_err(s1, "__bound_exit");
75 bound_init();
76 ret = (*prog_main)(argc, argv);
77 bound_exit();
78 return ret;
79 }
80#endif
81 return (*prog_main)(argc, argv);
82}

84/* relocate code. Return -1 on error, required size if ptr is NULL,
85 otherwise copy code into buffer passed by the caller */
86static int tcc_relocate_ex(TCCState *s1, void *ptr)
87{
88 Section *s;
89 unsigned long offset, length;
90 uplong mem;
91 int i;
92
93 if (0 == s1->runtime_added) {
94 s1->runtime_added = 1;
95 s1->nb_errors = 0;
96#ifdef TCC_TARGET_PE
97 pe_output_file(s1, NULL);
98#else
99 tcc_add_runtime(s1);
100 relocate_common_syms();
101 tcc_add_linker_symbols(s1);
102 build_got_entries(s1);
103#endif
104 if (s1->nb_errors)
105 return -1;
106 }
107
108 offset = 0, mem = (uplong) ptr;
109 for (i = 1; i < s1->nb_sections; i++) {
110 s = s1->sections[i];
111 if (0 == (s->sh_flags & SHF_ALLOC))
112 continue;
113 length = s->data_offset;
114 s->sh_addr = mem ? (mem + offset + 15) & ~15 : 0;
115 offset = (offset + length + 15) & ~15;
116 }
117 offset += 16;
118
119 /* relocate symbols */
120 relocate_syms(s1, 1);
121 if (s1->nb_errors)
122 return -1;
123
124#if defined TCC_TARGET_X86_64 && !defined TCC_TARGET_PE
125 s1->runtime_plt_and_got_offset = 0;
126 s1->runtime_plt_and_got = (char *) (mem + offset);
127 /* double the size of the buffer for got and plt entries
128 XXX: calculate exact size for them? */
129 offset *= 2;
130#endif
131
132 if (0 == mem)
133 return offset;
134
135 /* relocate each section */
136 for (i = 1; i < s1->nb_sections; i++) {
137 s = s1->sections[i];
138 if (s->reloc)
139 relocate_section(s1, s);
140 }
141
142 for (i = 1; i < s1->nb_sections; i++) {
143 s = s1->sections[i];
144 if (0 == (s->sh_flags & SHF_ALLOC))
145 continue;
146 length = s->data_offset;
147 // printf("%-12s %08x %04x\n", s->name, s->sh_addr, length);
148 ptr = (void *) (uplong) s->sh_addr;
149 if (NULL == s->data || s->sh_type == SHT_NOBITS)
150 memset(ptr, 0, length);
151 else
152 memcpy(ptr, s->data, length);
153 /* mark executable sections as executable in memory */
154 if (s->sh_flags & SHF_EXECINSTR)
155 set_pages_executable(ptr, length);
156 }
157
158#if defined TCC_TARGET_X86_64 && !defined TCC_TARGET_PE
159 set_pages_executable(s1->runtime_plt_and_got, s1->runtime_plt_and_got_offset);
160#endif
161 return 0;
162}

165/* allow to run code in memory */
166
167static void set_pages_executable(void *ptr, unsigned long length)
168{
169#ifdef _WIN32
170 unsigned long old_protect;
171 VirtualProtect(ptr, length, PAGE_EXECUTE_READWRITE, &old_protect);
172#else
173 unsigned long start, end;
174 start = (unsigned long) ptr & ~(PAGESIZE - 1);
175 end = (unsigned long) ptr + length;
176 end = (end + PAGESIZE - 1) & ~(PAGESIZE - 1);
177 mprotect((void *) start, end - start, PROT_READ | PROT_WRITE | PROT_EXEC);
178#endif
179}

184/* print the position in the source file of PC value 'pc' by reading
185 the stabs debug information */
186static unsigned long rt_printline(unsigned long wanted_pc)
187{
188 Stab_Sym *sym, *sym_end;
189 char func_name[128], last_func_name[128];
190 unsigned long func_addr, last_pc, pc;
191 const char *incl_files[INCLUDE_STACK_SIZE];
192 int incl_index, len, last_line_num, i;
193 const char *str, *p;
194
195 fprintf(stderr, "0x%08lx:", wanted_pc);
196
197 func_name[0] = '\0';
198 func_addr = 0;
199 incl_index = 0;
200 last_func_name[0] = '\0';
201 last_pc = 0xffffffff;
202 last_line_num = 1;
203 sym = (Stab_Sym *) stab_section->data + 1;
204 sym_end = (Stab_Sym *) (stab_section->data + stab_section->data_offset);
205 while (sym < sym_end) {
206 switch (sym->n_type) {
207 /* function start or end */
208 case N_FUN:
209 if (sym->n_strx == 0) {
210 /* we test if between last line and end of function */
211 pc = sym->n_value + func_addr;
212 if (wanted_pc >= last_pc && wanted_pc < pc)
213 goto found;
214 func_name[0] = '\0';
215 func_addr = 0;
216 } else {
217 str = stabstr_section->data + sym->n_strx;
218 p = strchr(str, ':');
219 if (!p) {
220 pstrcpy(func_name, sizeof(func_name), str);
221 } else {
222 len = p - str;
223 if (len > sizeof(func_name) - 1)
224 len = sizeof(func_name) - 1;
225 memcpy(func_name, str, len);
226 func_name[len] = '\0';
227 }
228 func_addr = sym->n_value;
229 }
230 break;
231 /* line number info */
232 case N_SLINE:
233 pc = sym->n_value + func_addr;
234 if (wanted_pc >= last_pc && wanted_pc < pc)
235 goto found;
236 last_pc = pc;
237 last_line_num = sym->n_desc;
238 /* XXX: slow! */
239 strcpy(last_func_name, func_name);
240 break;
241 /* include files */
242 case N_BINCL:
243 str = stabstr_section->data + sym->n_strx;
244 add_incl:
245 if (incl_index < INCLUDE_STACK_SIZE) {
246 incl_files[incl_index++] = str;
247 }
248 break;
249 case N_EINCL:
250 if (incl_index > 1)
251 incl_index--;
252 break;
253 case N_SO:
254 if (sym->n_strx == 0) {
255 incl_index = 0; /* end of translation unit */
256 } else {
257 str = stabstr_section->data + sym->n_strx;
258 /* do not add path */
259 len = strlen(str);
260 if (len > 0 && str[len - 1] != '/')
261 goto add_incl;
262 }
263 break;
264 }
265 sym++;
266 }
267
268 /* second pass: we try symtab symbols (no line number info) */
269 incl_index = 0;
270 {
271 ElfW(Sym) * sym, *sym_end;
272 int type;
273
274 sym_end = (ElfW(Sym) *) (symtab_section->data + symtab_section->data_offset);
275 for (sym = (ElfW(Sym) *) symtab_section->data + 1; sym < sym_end; sym++) {
276 type = ELFW(ST_TYPE)(sym->st_info);
277 if (type == STT_FUNC) {
278 if (wanted_pc >= sym->st_value && wanted_pc < sym->st_value + sym->st_size) {
279 pstrcpy(last_func_name,
280 sizeof(last_func_name),
281 strtab_section->data + sym->st_name);
282 func_addr = sym->st_value;
283 goto found;
284 }
285 }
286 }
287 }
288 /* did not find any info: */
289 fprintf(stderr, " ???\n");
290 return 0;
291found:
292 if (last_func_name[0] != '\0') {
293 fprintf(stderr, " %s()", last_func_name);
294 }
295 if (incl_index > 0) {
296 fprintf(stderr, " (%s:%d", incl_files[incl_index - 1], last_line_num);
297 for (i = incl_index - 2; i >= 0; i--)
298 fprintf(stderr, ", included from %s", incl_files[i]);
299 fprintf(stderr, ")");
300 }
301 fprintf(stderr, "\n");
302 return func_addr;
303}

305/* emit a run time error at position 'pc' */
306static void rt_error(ucontext_t *uc, const char *fmt, ...)
307{
308 va_list ap;
309 unsigned long pc;
310 int i;
311
312 va_start(ap, fmt);
313 fprintf(stderr, "Runtime error: ");
314 vfprintf(stderr, fmt, ap);
315 fprintf(stderr, "\n");
316
317 for (i = 0; i < num_callers; i++) {
318 if (rt_get_caller_pc(&pc, uc, i) < 0)
319 break;
320 if (i == 0)
321 fprintf(stderr, "at ");
322 else
323 fprintf(stderr, "by ");
324 pc = rt_printline(pc);
325 if (pc == (unsigned long) rt_prog_main && pc)
326 break;
327 }
328 exit(255);
329 va_end(ap);
330}

335/* signal handler for fatal errors */
336static void sig_error(int signum, siginfo_t *siginf, void *puc)
337{
338 ucontext_t *uc = puc;
339
340 switch (signum) {
341 case SIGFPE:
342 switch (siginf->si_code) {
343 case FPE_INTDIV:
344 case FPE_FLTDIV:
345 rt_error(uc, "division by zero");
346 break;
347 default:
348 rt_error(uc, "floating point exception");
349 break;
350 }
351 break;
352 case SIGBUS:
353 case SIGSEGV:
354 if (rt_bound_error_msg && *rt_bound_error_msg)
355 rt_error(uc, *rt_bound_error_msg);
356 else
357 rt_error(uc, "dereferencing invalid pointer");
358 break;
359 case SIGILL:
360 rt_error(uc, "illegal instruction");
361 break;
362 case SIGABRT:
363 rt_error(uc, "abort() called");
364 break;
365 default:
366 rt_error(uc, "caught signal %d", signum);
367 break;
368 }
369 exit(255);
370}

372/* Generate a stack backtrace when a CPU exception occurs. */
373static void set_exception_handler(void)
374{
375 struct sigaction sigact;
376 /* install TCC signal handlers to print debug info on fatal
377 runtime errors */
378 sigact.sa_flags = SA_SIGINFO | SA_RESETHAND;
379 sigact.sa_sigaction = sig_error;
380 sigemptyset(&sigact.sa_mask);
381 sigaction(SIGFPE, &sigact, NULL);
382 sigaction(SIGILL, &sigact, NULL);
383 sigaction(SIGSEGV, &sigact, NULL);
384 sigaction(SIGBUS, &sigact, NULL);
385 sigaction(SIGABRT, &sigact, NULL);
386}

397/* return the PC at frame level 'level'. Return non zero if not found */
398static int rt_get_caller_pc(unsigned long *paddr, ucontext_t *uc, int level)
399{
400 unsigned long fp;
401 int i;
402
403 if (level == 0) {
404#if defined(__FreeBSD__)
405 *paddr = uc->uc_mcontext.mc_eip;
406#elif defined(__dietlibc__)
407 *paddr = uc->uc_mcontext.eip;
408#else
409 *paddr = uc->uc_mcontext.gregs[REG_EIP];
410#endif
411 return 0;
412 } else {
413#if defined(__FreeBSD__)
414 fp = uc->uc_mcontext.mc_ebp;
415#elif defined(__dietlibc__)
416 fp = uc->uc_mcontext.ebp;
417#else
418 fp = uc->uc_mcontext.gregs[REG_EBP];
419#endif
420 for (i = 1; i < level; i++) {
421 /* XXX: check address validity with program info */
422 if (fp <= 0x1000 || fp >= 0xc0000000)
423 return -1;
424 fp = ((unsigned long *) fp)[0];
425 }
426 *paddr = ((unsigned long *) fp)[1];
427 return 0;
428 }
429}
