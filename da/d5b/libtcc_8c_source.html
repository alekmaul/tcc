428/* copy a string and truncate it. */
429char *pstrcpy(char *buf, int buf_size, const char *s)
430{
431 char *q, *q_end;
432 int c;
433
434 if (buf_size > 0) {
435 q = buf;
436 q_end = buf + buf_size - 1;
437 while (q < q_end) {
438 c = *s++;
439 if (c == '\0')
440 break;
441 *q++ = c;
442 }
443 *q = '\0';
444 }
445 return buf;
446}

546void dynarray_add(void ***ptab, int *nb_ptr, void *data)
547{
548 int nb, nb_alloc;
549 void **pp;
550
551 nb = *nb_ptr;
552 pp = *ptab;
553 /* every power of two we double array size */
554 if ((nb & (nb - 1)) == 0) {
555 if (!nb)
556 nb_alloc = 1;
557 else
558 nb_alloc = nb * 2;
559 pp = tcc_realloc(pp, nb_alloc * sizeof(void *));
560 *ptab = pp;
561 }
562 pp[nb++] = data;
563 *nb_ptr = nb;
564}

576/* symbol allocator */
577static Sym *__sym_malloc(void)
578{
579 Sym *sym_pool, *sym, *last_sym;
580 int i;
581
582 sym_pool = tcc_malloc(SYM_POOL_NB * sizeof(Sym));
583 dynarray_add(&sym_pools, &nb_sym_pools, sym_pool);
584
585 last_sym = sym_free_first;
586 sym = sym_pool;
587 for (i = 0; i < SYM_POOL_NB; i++) {
588 sym->next = last_sym;
589 last_sym = sym;
590 sym++;
591 }
592 sym_free_first = last_sym;
593 return last_sym;
594}

612Section *new_section(TCCState *s1, const char *name, int sh_type, int sh_flags)
613{
614 Section *sec;
615
616 sec = tcc_mallocz(sizeof(Section) + strlen(name));
617 strcpy(sec->name, name);
618 sec->sh_type = sh_type;
619 sec->sh_flags = sh_flags;
620 switch (sh_type) {
621 case SHT_HASH:
622 case SHT_REL:
623 case SHT_RELA:
624 case SHT_DYNSYM:
625 case SHT_SYMTAB:
626 case SHT_DYNAMIC:
627 sec->sh_addralign = 4;
628 break;
629 case SHT_STRTAB:
630 sec->sh_addralign = 1;
631 break;
632 default:
633 sec->sh_addralign = 32; /* default conservative alignment */
634 break;
635 }
636
637 if (sh_flags & SHF_PRIVATE) {
638 dynarray_add((void ***) &s1->priv_sections, &s1->nb_priv_sections, sec);
639 } else {
640 sec->sh_num = s1->nb_sections;
641 dynarray_add((void ***) &s1->sections, &s1->nb_sections, sec);
642 }
643
644 return sec;
645}

652/* realloc section and set its content to zero */
653static void section_realloc(Section *sec, unsigned long new_size)
654{
655 unsigned long size;
656 unsigned char *data;
657
658 size = sec->data_allocated;
659 if (size == 0)
660 size = 1;
661 while (size < new_size)
662 size = size * 2;
663 data = tcc_realloc(sec->data, size);
664 memset(data + sec->data_allocated, 0, size - sec->data_allocated);
665 sec->data = data;
666 sec->data_allocated = size;
667}

669/* reserve at least 'size' bytes in section 'sec' from
670 sec->data_offset. */
671static void *section_ptr_add(Section *sec, unsigned long size)
672{
673 unsigned long offset, offset1;
674
675 offset = sec->data_offset;
676 offset1 = offset + size;
677 if (offset1 > sec->data_allocated)
678 section_realloc(sec, offset1);
679 sec->data_offset = offset1;
680 return sec->data + offset;
681}

744#ifdef CONFIG_TCC_BCHECK

746 char buf[32];

755 case TOK_free:

772#endif

774#ifdef TCC_TARGET_PE

776 sprintf(buf1, "_%s@%d", name, FUNC_ARGS(attr));
777 name = buf1;
778 } else
779#endif
780 if (tcc_state->leading_underscore && can_add_underscore) {
781 buf1[0] = '_';
782 pstrcpy(buf1 + 1, sizeof(buf1) - 1, name);
783 name = buf1;
784 }
785#ifdef TCC_TARGET_816
786 if (sym->type.t & VT_STATIC) {
787 if ((sym->type.t & VT_STATICLOCAL) && current_fn[0] != 0)
788 sprintf(buf1, "%s_FUNC_%s_", STATIC_PREFIX, current_fn);

1083/* push a given symbol on the symbol stack */
1084static Sym *sym_push(int v, CType *type, int r, int c)
1085{
1086 Sym *s, **ps;
1087 TokenSym *ts;
1088
1089 if (local_stack)
1090 ps = &local_stack;
1091 else
1092 ps = &global_stack;
1093 s = sym_push2(ps, v, type->t, c);
1094 s->type.ref = type->ref;
1095 s->r = r;
1096 /* don't record fields or anonymous symbols */
1097 /* XXX: simplify */
1098 if (!(v & SYM_FIELD) && (v & ~SYM_STRUCT) < SYM_FIRST_ANOM) {
1099 /* record symbol in token array */
1100 ts = table_ident[(v & ~SYM_STRUCT) - TOK_IDENT];
1101 if (v & SYM_STRUCT)
1102 ps = &ts->sym_struct;
1103 else
1104 ps = &ts->sym_identifier;
1105 s->prev_tok = *ps;
1106 *ps = s;
1107 }
1108 return s;
1109}

1111/* push a global identifier */
1112static Sym *global_identifier_push(int v, int t, int c)
1113{
1114 Sym *s, **ps;
1115 s = sym_push2(&global_stack, v, t, c);
1116 /* don't record anonymous symbol */
1117 if (v < SYM_FIRST_ANOM) {
1118 ps = &table_ident[v - TOK_IDENT]->sym_identifier;
1119 /* modify the top most local identifier, so that
1120 sym_identifier will point to 's' when popped */
1121 while (*ps != NULL)
1122 ps = &(*ps)->prev_tok;
1123 s->prev_tok = NULL;
1124 *ps = s;
1125 }
1126 return s;
1127}

1129/* pop symbols until top reaches 'b' */
1130static void sym_pop(Sym **ptop, Sym *b)
1131{
1132 Sym *s, *ss, **ps;
1133 TokenSym *ts;
1134 int v;
1135
1136 s = *ptop;
1137 while (s != b) {
1138 ss = s->prev;
1139 v = s->v;
1140 /* remove symbol in token array */
1141 /* XXX: simplify */
1142 if (!(v & SYM_FIELD) && (v & ~SYM_STRUCT) < SYM_FIRST_ANOM) {
1143 ts = table_ident[(v & ~SYM_STRUCT) - TOK_IDENT];
1144 if (v & SYM_STRUCT)
1145 ps = &ts->sym_struct;
1146 else
1147 ps = &ts->sym_identifier;
1148 *ps = s->prev_tok;
1149 }
1150 sym_free(s);
1151 s = ss;
1152 }
1153 *ptop = b;
1154}

1204
1205#ifdef INC_DEBUG
1206 printf("%s: **** new file\n", file->filename);
1207#endif
1208 preprocess_init(s1);
1209
1210 cur_text_section = NULL;
1211 funcname = "";
1212 anon_sym = SYM_FIRST_ANOM;
1213
1214 /* file info: full path + filename */
1215 section_sym = 0; /* avoid warning */
1216 if (s1->do_debug) {
1217 section_sym = put_elf_sym(symtab_section,
1218 0,
1219 0,
1220 ELFW(ST_INFO)(STB_LOCAL, STT_SECTION),
1221 0,
1222 text_section->sh_num,
1223 NULL);

1225#ifdef _WIN32

1230 put_stabs_r(file->filename, N_SO, 0, 0, text_section->data_offset, text_section, section_sym);
1231 }

1247#else

1249#endif

1253

1257

1261 func_double_type.ref = sym_push(SYM_FIELD, &double_type, FUNC_CDECL, FUNC_OLD);
1262#endif

1264#if 0
1265 /* define 'void *alloca(unsigned int)' builtin function */
1266 {
1267 Sym *s1;
1268
1269 p = anon_sym++;
1270 sym = sym_push(p, mk_pointer(VT_VOID), FUNC_CDECL, FUNC_NEW);
1271 s1 = sym_push(SYM_FIELD, VT_UNSIGNED | VT_INT, 0, 0);

1273 sym->next = s1;

1276#endif
1277
1278 define_start = define_stack;
1279 nocode_wanted = 1;
1280
1281 if (setjmp(s1->error_jmp_buf) == 0) {
1282 s1->nb_errors = 0;
1283 s1->error_set_jmp_enabled = 1;
1284
1285 ch = file->buf_ptr[0];

1287 parse_flags = PARSE_FLAG_PREPROCESS | PARSE_FLAG_TOK_NUM;
1288 next();
1289 decl(VT_CONST);
1290 if (tok != TOK_EOF)
1291 expect("declaration");
1292
1293 /* end of translation unit info */
1294 if (s1->do_debug) {
1295 put_stabs_r(NULL, N_SO, 0, 0, text_section->data_offset, text_section, section_sym);
1296 }
1297 }
1298 s1->error_set_jmp_enabled = 0;
1299
1300 /* reset define stack, but leave -Dsymbols (may be incorrect if
1301 they are undefined) */
1302 free_defines(define_start);
1303
1304 gen_inline_functions();
1305
1306#ifndef TCC_TARGET_816
1307 sym_pop(&global_stack, NULL);
1308 sym_pop(&local_stack, NULL);
1309#endif
1310
1311 return s1->nb_errors != 0 ? -1 : 0;
1312}

1615static int tcc_add_file_internal(TCCState *s1, const char *filename, int flags)
1616{
1617 const char *ext;
1618#ifdef TCC_TARGET_816
1619 int ret;
1620#else
1621 ElfW(Ehdr) ehdr;
1622 int fd, ret, size;
1623#endif
1624 BufferedFile *saved_file;
1625
1626 ret = -1;
1627
1628 /* find source file type with extension */
1629 ext = tcc_fileextension(filename);
1630 if (ext[0])
1631 ext++;
1632
1633 /* open the file */
1634 saved_file = file;
1635 file = tcc_open(s1, filename);
1636 if (!file) {
1637 if (flags & AFF_PRINT_ERROR)
1638 error_noabort("file '%s' not found", filename);
1639 goto the_end;
1640 }
1641
1642 if (flags & AFF_PREPROCESS) {
1643 ret = tcc_preprocess(s1);
1644 goto the_end;
1645 }
1646
1647 if (!ext[0] || !PATHCMP(ext, "c")) {
1648 /* C file assumed */
1649 ret = tcc_compile(s1);
1650 goto the_end;
1651 }
1652
1653#ifdef CONFIG_TCC_ASM
1654 if (!strcmp(ext, "S")) {
1655 /* preprocessed assembler */
1656 ret = tcc_assemble(s1, 1);
1657 goto the_end;
1658 }
1659
1660 if (!strcmp(ext, "s")) {
1661 /* non preprocessed assembler */
1662 ret = tcc_assemble(s1, 0);
1663 goto the_end;
1664 }
1665#endif
1666
1667#ifdef TCC_TARGET_816
1668 error_noabort("unrecognized file type");
1669 goto the_end;
1670#else
1671 fd = file->fd;
1672 /* assume executable format: auto guess file type */
1673 size = read(fd, &ehdr, sizeof(ehdr));
1674 lseek(fd, 0, SEEK_SET);
1675 if (size <= 0) {
1676 error_noabort("could not read header");
1677 goto the_end;
1678 }
1679
1680 if (size == sizeof(ehdr) && ehdr.e_ident[0] == ELFMAG0 && ehdr.e_ident[1] == ELFMAG1
1681 && ehdr.e_ident[2] == ELFMAG2 && ehdr.e_ident[3] == ELFMAG3) {
1682 /* do not display line number if error */
1683 file->line_num = 0;
1684 if (ehdr.e_type == ET_REL) {
1685 ret = tcc_load_object_file(s1, fd, 0);
1686 goto the_end;
1687 }
1688#ifndef TCC_TARGET_PE
1689 if (ehdr.e_type == ET_DYN) {
1690 if (s1->output_type == TCC_OUTPUT_MEMORY) {
1691 void *h;
1692 h = dlopen(filename, RTLD_GLOBAL | RTLD_LAZY);
1693 if (h)
1694 ret = 0;
1695 } else {
1696 ret = tcc_load_dll(s1, fd, filename, (flags & AFF_REFERENCED_DLL) != 0);
1697 }
1698 goto the_end;
1699 }
1700#endif
1701 error_noabort("unrecognized ELF file");
1702 goto the_end;
1703 }
1704
1705 if (memcmp((char *) &ehdr, ARMAG, 8) == 0) {
1706 file->line_num = 0; /* do not display line number if error */
1707 ret = tcc_load_archive(s1, fd);
1708 goto the_end;
1709 }
1710
1711#ifdef TCC_TARGET_COFF
1712 if (*(uint16_t *) (&ehdr) == COFF_C67_MAGIC) {
1713 ret = tcc_load_coff(s1, fd);
1714 goto the_end;
1715 }
1716#endif
1717
1718#ifdef TCC_TARGET_PE
1719 ret = pe_load_file(s1, filename, fd);
1720#else
1721 /* as GNU ld, consider it is an ld script if not recognized */
1722 ret = tcc_load_ldscript(s1);
1723#endif
1724 if (ret < 0)
1725 error_noabort("unrecognized file type");
1726#endif
1727the_end:
1728 if (file)
1729 tcc_close(file);
1730 file = saved_file;

1732}
